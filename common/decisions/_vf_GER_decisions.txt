### オーストリア帝冠領の再編 ###
vf_GER_crownland_reorganization = {
	# ボヘミア地位法
	vf_GER_decision_bohemia_statute = {
		icon = generic_civil_support

		fire_only_once = yes

		cost = var:vf_GER_decision_cost_75

		available = {
			country_exists = BOH
		}

		complete_effect = {
			set_country_flag = vf_GER_bohemia_statute
			# BOHの自治度を上げ、忠誠フラグを立てる（指示書 §8-3）
			BOH = {
				add_autonomy_ratio = {
					value = 0.05
					localization = vf_autonomy_crownland_statute
				}
			}
			hidden_effect = {
				set_temp_variable = {
					vf_temp_bop_regent = -10
				}
				vf_GER_change_power_balance = yes
				country_event = {
					id = event_GER_pol.16
					days = 3
				}
			}
		}
	}
	# クロアチア地位法
	vf_GER_decision_croatia_statute = {
		icon = generic_civil_support

		fire_only_once = yes

		cost = var:vf_GER_decision_cost_75

		available = {
			country_exists = CRO
		}

		complete_effect = {
			# 防衛条項フラグ（指示書 §8-3）
			set_country_flag = vf_GER_croatia_defense_clause
			CRO = {
				add_autonomy_ratio = {
					value = 0.05
					localization = vf_autonomy_crownland_statute
				}
			}
			hidden_effect = {
				set_temp_variable = {
					vf_temp_bop_regent = -10
				}
				vf_GER_change_power_balance = yes
				country_event = {
					id = event_GER_pol.16
					days = 3
				}
			}
		}
	}
	# 帝冠領財政の帝国予算統合
	vf_GER_decision_crownland_budget_integration = {
		icon = generic_form_nation

		fire_only_once = yes

		cost = var:vf_GER_decision_cost_50

		complete_effect = {
			set_country_flag = vf_GER_crownland_budget_done
			# TODO(VF): 効果の詳細はloreに無いため仮置き（首都に民需工場+1）。主宰へ報告
			capital_scope = {
				add_building_construction = {
					type = industrial_complex
					level = 1
				}
			}
		}
	}
	# 墺系諸邦の行政通則化
	vf_GER_decision_austrian_administration = {
		icon = generic_form_nation

		fire_only_once = yes

		cost = var:vf_GER_decision_cost_50

		complete_effect = {
			set_country_flag = vf_GER_crownland_admin_done
			add_stability = 0.02
			hidden_effect = {
				set_temp_variable = {
					vf_temp_bop_regent = -5
				}
				vf_GER_change_power_balance = yes
			}
		}
	}
	# 帝冠領整理の完遂（上記4種の完了で解禁）
	vf_GER_decision_crownland_completion = {
		icon = generic_form_nation

		fire_only_once = yes

		cost = 0

		available = {
			has_country_flag = vf_GER_bohemia_statute
			has_country_flag = vf_GER_croatia_defense_clause
			has_country_flag = vf_GER_crownland_budget_done
			has_country_flag = vf_GER_crownland_admin_done
		}

		complete_effect = {
			set_country_flag = vf_GER_crownland_complete
			# 国民精神「未解決の遺産」を除去（指示書 §6-2）
			IF = {
				limit = {
					has_idea = idea_GER_unresolved_legacy
				}
				remove_ideas = idea_GER_unresolved_legacy
			}
			add_political_power = 50
		}
	}
}
