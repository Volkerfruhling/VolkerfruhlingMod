# =========================================================
# VF / 大国のAI師団設計図（仮）
#
# 【なぜこのファイルが必要か】
# バニラ由来の common/ai_templates/generic.txt は GER / FRA / ITA / RUS を blocked_for で除外している。
# バニラでは各国専用の ai_templates ファイルが別に存在するためだが、このModにはそれが無い。
# 結果としてこの4か国は「どの役割の設計図も持たない」状態になり、AIが師団を設計・更新・生産できない。
# （09 §7 の「開戦時にGERが44個師団しかいない」問題の主因。ENG/USA/JAPは除外されていないので generic を使えている）
#
# 【方針】
# βでは4か国共通の最小構成を与える。国ごとの作り分けは本リリースで行う（下の TODO(VF) 参照）。
# 開始時OOB（history/units/*_1925.txt）の編成に近い形を early に置き、AIが開幕から
# 大規模な再設計でXPを浪費しないようにしてある。
#
# 関連ファイル:
#   common/ai_templates/generic.txt         … ENG/USA/JAP等が使う汎用設計図。4か国を blocked_for で除外している
#   common/ai_strategy/_vf_GreatPowers.txt  … role_ratio（infantry/artillery）と ai_wanted_divisions_factor
#   history/units/GER_1925.txt 他            … 開始時OOBの division_template
#
# 参照する role: infantry / armor / mountaineer
#   ※ generic.txt に実在する role トークンのみを使用している（cavalry_generic と motorized_generic は
#     generic.txt 側で role = marine になっているが、バニラ由来のため本ファイルでは踏襲せず触らない）
# =========================================================

### 歩兵線（4大国共通） ###
vf_major_line_infantry = {
	available_for = {
		GER
		FRA
		ITA
		RUS
	}
	role = infantry
	upgrade_prio = {
		base = 100
	}

	# 1925年の標準師団。開始時OOBの「Infanterie-Division」「Russian Infantry Division」等に対応する
	vf_major_infantry_early = {
		upgrade_prio = {
			base = 10
		}
		can_upgrade_in_field = {
			has_equipment = {
				infantry_equipment > 100
			}
		}
		target_template = {
			support = {
				engineer = 1
				artillery = 1
			}
			regiments = {
				infantry = 9
			}
		}
		replace_at_match = 0.1
		target_min_match = 0.1
		replace_with = vf_major_infantry_upgraded
	}

	# 軍需工場が育ってきたら砲兵を組み込んだ編成へ移行する
	vf_major_infantry_upgraded = {
		upgrade_prio = {
			base = 0
			modifier = {
				add = 10000
				num_of_military_factories > 18
			}
		}
		can_upgrade_in_field = {
			has_equipment = {
				infantry_equipment > 100
			}
		}
		target_template = {
			support = {
				engineer = 1
				recon = 1
				artillery = 1
			}
			regiments = {
				infantry = 9
				artillery_brigade = 1
			}
		}
		replace_at_match = 0.1
		target_min_match = 0.1
	}

	# 認識用（優先度0）。開始時の騎兵旅団を歩兵線の一部としてAIに認識させ、
	# 更新の対象に含める。ここが無いと騎兵師団が「役割不明」のまま放置される
	vf_major_cavalry_legacy = {
		upgrade_prio = {
			base = 0
		}
		can_upgrade_in_field = {
			has_equipment = {
				infantry_equipment > 100
			}
		}
		target_template = {
			support = {
				recon = 1
				artillery = 1
			}
			regiments = {
				cavalry = 4
			}
		}
		replace_at_match = 0.1
		target_min_match = 0.1
	}
}
