# 議会ルート: 議会派のシェアが40%以上
vf_GER_bop_route_parliament_possible = {
	check_variable = {
		vf_GER_bop_assembly_share > 39.99
	}
}

# アウスグライヒ（君主）ルート: 皇帝派＋枢密院のスコアが50%以上
vf_GER_bop_route_monarch_possible = {
	check_variable = {
		vf_GER_bop_monarch_score > 49.99
	}
}

# 内戦ルート: どちらの条件も満たさない（拮抗）
vf_GER_bop_route_civilwar_possible = {
	NOT = {
		vf_GER_bop_route_parliament_possible = yes
	}
	NOT = {
		vf_GER_bop_route_monarch_possible = yes
	}
}
