# VF / 開戦前のAI集結制御
# 「宣戦したのに国境ががら空き／戦線が薄くて突破できない」を防ぐため、
# NFの裏効果（hidden_effect のグローバルフラグ）を読んで開戦前から国境へ軍を寄せる。
#   vf_SER_preparing_war_ALB … focus_SER_pol_lost_kosovo が立てる（平定作戦の2NF前）
#   vf_SER_preparing_war_BUL … focus_SER_pol_oppose_balkan_pact が立てる（復讐の日の1NF前）
#   vf_SER_preparing_war_CRO … focus_SER_pol_the_last_irredenta が立てる（開戦の3NF前）
VF_serbia_prep_albania_strategy = {
	allowed = {
		original_tag = SER
	}
	enable = {
		original_tag = SER
		has_global_flag = vf_SER_preparing_war_ALB
		country_exists = ALB
		NOT = {
			has_war_with = ALB
		}
	}
	abort_when_not_enabled = yes
	# 全軍をALB国境の戦線へ（他国境は薄くてよい）
	ai_strategy = {
		type = front_unit_request
		tag = ALB
		value = 300
	}
	ai_strategy = {
		type = prepare_for_war
		id = ALB
		value = 200
	}
	ai_strategy = {
		type = prepare_for_war
		id = BUL
		value = -100
	}
	ai_strategy = {
		type = prepare_for_war
		id = HUN
		value = -100
	}
}
