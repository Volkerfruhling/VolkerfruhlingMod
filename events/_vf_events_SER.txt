country_event = { # アルバニアの崩壊
	# 平定作戦の決着を保証する。ALBが押され始める（surrender_progress > 0.2）か、
	# 首都陥落、または開戦から120日経過のいずれかで政府が崩壊し、SERが併合する。
	# これにより「復讐の日」（available で対ALB戦争中は取得不可）が確実に解禁される
	id = event_SER_pol.12
	title = event_SER_pol.12.title
	desc = event_SER_pol.12.desc

	fire_only_once = yes

	trigger = {
		tag = SER
		has_war_with = ALB
		has_country_flag = vf_SER_albania_war
		OR = {
			ALB = {
				surrender_progress > 0.2
			}
			ALB = {
				has_capitulated = yes
			}
			# 泥沼化の保険: 開戦から120日で山賊国家は内部から崩壊する
			has_country_flag = {
				flag = vf_SER_albania_war
				days > 120
			}
		}
	}

	immediate = {
		SER = {
			annex_country = {
				target = ALB
			}
		}
	}

	option = {
		name = event_SER_pol.12.a
		add_war_support = 0.05
	}
}

country_event = { # 講話
	id = event_SER_pol.10
	title = event_SER_pol.10.title
	desc = event_SER_pol.10.desc

	fire_only_once = yes 

	trigger = {
		# tag = SER: 旧実装は国指定が無く、任意のAI国のスコープで発火していた（指示書 §4-0-1）
		tag = SER
		has_global_flag = vf_started_balkan_war
		NOT = {
			has_global_flag = vf_ended_balkan_war
		}
		OR = {
			# 通常講和: 開戦330日経過（.11 が立てる）かつBULが押されている
			AND = {
				has_global_flag = vf_balkan_war_min_duration_passed
				BUL = {
					surrender_progress > 0
				}
			}
			# BULが完全降伏していれば日数を待たない（指示書 §4-0-2）
			BUL = {
				has_capitulated = yes
			}
		}
	}
	immediate = {
		set_global_flag = vf_ended_balkan_war
		BUL = {
			leave_faction = yes
		}
		HUN = {
			leave_faction = yes
		}
		# 講和段階フラグ vf_SER_peace_tier_1〜4 を立てる（1が最大の戦果）。
		# 幕間枝（focus_SER_pol_after_the_peace 以降）がこの結果を読む（指示書 §6-4）
		IF = {
			limit = {
				BUL = {
					surrender_progress > 0.7
				}
			}
			set_global_flag = vf_SER_peace_tier_1
			BUL = {
				white_peace = {
					tag = SER
				}
				803 = {
					transfer_state_to = SER
					add_core_of = SER
					remove_core_of = BUL
					add_claim_by = BUL
				}
				106 = {
					transfer_state_to = SER
					add_core_of = SER
					remove_core_of = BUL
					add_claim_by = BUL
				}
				77 = {
					transfer_state_to = ROM
					add_core_of = ROM
					remove_core_of = BUL
					add_claim_by = BUL
				}
				184 = {
					transfer_state_to = GRE
					add_core_of = GRE
					remove_core_of = BUL
					add_claim_by = BUL
				}

			}
		}
		ELSE_IF = {
			limit = {
				BUL = {
					surrender_progress > 0.5
				}
			}
			set_global_flag = vf_SER_peace_tier_2
			BUL = {
				white_peace = {
					tag = SER
				}
				803 = {
					transfer_state_to = SER
					add_core_of = SER
					remove_core_of = BUL
					add_claim_by = BUL
				}
				106 = {
					transfer_state_to = SER
					add_core_of = SER
					remove_core_of = BUL
					add_claim_by = BUL
				}
			}
		}
		ELSE_IF = {
			limit = {
				BUL = {
					surrender_progress > 0.3
				}
			}
			set_global_flag = vf_SER_peace_tier_3
			BUL = {
				white_peace = {
					tag = SER
				}
				803 = {
					transfer_state_to = SER
					add_core_of = SER
					remove_core_of = BUL
					add_claim_by = BUL
				}
			}
		}
		ELSE = {
			# 白紙和平。「またしても何も得られなかった」（幕間枝 #11 が安定度ペナルティを課す）
			set_global_flag = vf_SER_peace_tier_4
			BUL = {
				white_peace = {
					tag = SER
				}
			}
		}
	}

	option = {
		name = event_SER_pol.10.a
	}
}
